// 1. 먼저 Animal 클래스를 만든다. 이때 속성은 종류, 자신의 번호, 방문 여부, 폐쇄 여부로 한다.  
enum Type{  
    SHEEP,  
    WOLF  
}  
class Animal {  
    Type type;  
    int number;  
    boolean visited;  
    boolean closed;  
    List<Animal> children;  
  
    public Animal(Type type, int number) {  
        this.type = type;  
        this.number = number;  
        this.children = new ArrayList<>();  
    }  
  
    public void updateVisited() {  
        this.visited = true;  
    }  
    public void updateClosed() {  
        this.closed = true;  
    }  
  
    @Override  
    public String toString() {  
        return "Animal{" +  
                "type=" + type +  
                ", number=" + number +  
                ", visited=" + visited +  
                ", closed=" + closed +  
                ", children= "+ children.stream().map(animal -> animal.number).toList()+ "}\n";  
    }  
}  
  
// 2. Owner 클래스를 만들고 속성은 양 마릿수, 늑대 마릿수, 자신의 위치로 한다.  
class Owner {  
    int sheepNum;  
    int wolfNum;  
  
    @Override  
    public String toString() {  
        return "Owner{" +  
                "sheepNum=" + sheepNum +  
                ", wolfNum=" + wolfNum +  
                '}';  
    }  
}  
  
// 5. Path 클래스를 만든다. 속성은 가는 루트 String, 늑대수, 양의 수로 한다.  
class Path{  
    ArrayDeque<Animal> queue;  
    int sheepNum;  
    int wolfNum;  
  
    Path(){  
        this.queue = new ArrayDeque<>();  
    }  
  
    public static Path copyPath(Path path){  
        Path newPath = new Path();  
        newPath.queue = new ArrayDeque<>(path.queue);  
        newPath.sheepNum = path.sheepNum;  
        newPath.wolfNum = path.wolfNum;  
        return newPath;  
    }  
  
    @Override  
    public String toString() {  
        return "Path : " + queue.stream().map(animal -> animal.number).toList() +  
                " , sheepNum : " + sheepNum + " " +  
                ", wolfNum : " + wolfNum + "";  
    }  
}  
  
class Solution {  
    public int solution(int[] info, int[][] edges) {  
        // 1. 먼저 Animal 클래스를 만든다. 이때 속성은 종류, 자신의 번호, 방문 여부, 폐쇄 여부로 한다.  
        int infoLen = info.length;  
        Map<Integer, Animal> animalMap = new HashMap<>();  
        for(int i = 0; i < infoLen; i++) {  
            Type type = info[i] == 0 ? Type.SHEEP : Type.WOLF;  
            animalMap.put(i, new Animal(type, i));  
        }  
  
        // 3. Tree는 edge를 통해 부모와 자식간의 관계를 서로 이어준다. 즉, Animal 에 children 속성을 추가한다.  
        int edgeLen = edges.length;  
        for(int i = 0; i < edgeLen; i++) {  
            Animal parent = animalMap.get(edges[i][0]);  
            Animal child = animalMap.get(edges[i][1]);  
            parent.children.add(child);  
        }  
  
        // 2. Owner 클래스를 만들고 속성은 양 마릿수, 늑대 마릿수, 자신의 위치로 한다.  
        ArrayList<Path> pathTree = new ArrayList<>();  
        Owner owner = new Owner();  
        owner.sheepNum++;  
        animalMap.get(0).updateVisited();  
        move(animalMap.get(0), owner, pathTree);  
  
        return owner.sheepNum;  
    }  
  
    // 먼저 양들부터 먹어준다.  
    private void move(Animal currentPos, Owner owner, ArrayList<Path> pathTree){  
        if(currentPos == null) return;  
        if(currentPos.children.isEmpty()) return;  
  
        System.out.println("currentPos : " + currentPos);  
        System.out.println("owner : " + owner);  
        currentPos.children.forEach(animal -> {  
            if(animal != null && animal.type == Type.SHEEP) {  
                visitSheep(animal, owner);  
                move(animal, owner, pathTree);  
            }  
        });  
  
        currentPos.children.forEach(animal -> {  
            if(animal != null && animal.type == Type.WOLF){  
                Path path = new Path();  
                path.queue.add(animal);  
                path.wolfNum++;  
                getPath(animal, path, owner.sheepNum - owner.wolfNum, 1, pathTree);  
  
                List<Path> sort = pathTree.stream()  
                        .sorted(Comparator.comparing(p -> p.wolfNum - p.sheepNum)).toList();  
  
                Animal pathAnimal = null;  
                for(Path p: sort){  
                    while(!p.queue.isEmpty()){  
                        pathAnimal = p.queue.poll();  
                        if(pathAnimal.type == Type.SHEEP){  
                            visitSheep(pathAnimal, owner);  
                        }else{  
                            visitWolf(pathAnimal, owner);  
                        }  
                    }  
                    move(pathAnimal, owner, pathTree);  
                }  
            }  
        });  
    }  
  
    // Path 를 limit 만큼 한개 얻는다.  
    private void getPath(Animal baseAnimal, Path currentPath, int limit, int num, ArrayList<Path> pathTree){  
        if(num >= limit || baseAnimal.children.isEmpty()){  
            // 넣기 전에 끝 늑대들 절단  
            while(currentPath.queue.getLast().type == Type.WOLF && currentPath.queue.getLast().children.isEmpty()){  
                currentPath.queue.removeLast();  
                currentPath.wolfNum--;  
  
                if(currentPath.queue.isEmpty()) break;  
            }  
            if(!currentPath.queue.isEmpty()){  
                pathTree.add(currentPath);  
                System.out.println("currentPath : " + currentPath);  
            };  
  
            return;  
        }  
  
        for (Animal animal : baseAnimal.children) {  
            if(animal != null){  
                Path newPath = Path.copyPath(currentPath);  
                newPath.queue.add(animal);  
                num++;  
  
                if(animal.type == Type.SHEEP){  
                    newPath.sheepNum++;  
                }else{  
                    newPath.wolfNum++;  
                }  
                getPath(animal, newPath, limit, num, pathTree);  
            }  
        }  
    }  
  
    private void visitSheep(Animal currentPos, Owner owner){  
        if(!currentPos.visited){  
            owner.sheepNum++;  
            currentPos.updateVisited();  
        }  
    }  
  
    private void visitWolf(Animal currentPos, Owner owner){  
        if(!currentPos.visited){  
            owner.wolfNum++;  
            currentPos.updateVisited();  
        }  
    }  
}
